        const [searchQ, setSearchQ] = useState('');

        const myStaffRecord = staffData.find(s => s.id === auth?.username);

        const handleStatusChange = (newStatus) => {
            const updated = staffData.map(s => s.id === auth?.username ? { ...s, status: newStatus } : s);
            localStorage.setItem('cf_staff', JSON.stringify(updated));
            setStaffData(updated);

            const acts = JSON.parse(localStorage.getItem('cf_activity')||'[]');
            acts.unshift({ id: Date.now(), text: `${auth?.name} changed status to ${newStatus}`, time: new Date().toISOString() });
            localStorage.setItem('cf_activity', JSON.stringify(acts));
        };

        const filteredStaff = staffData.filter(s => {
            if (filterDept !== 'All Departments' && s.department !== filterDept) return false;
            if (filterStatus !== 'All Status' && s.status !== filterStatus) return false;
            if (searchQ && !s.name.toLowerCase().includes(searchQ.toLowerCase())) return false;
            return true;
        });

        const statusColors = {
            'AVAILABLE': 'bg-emerald-500',
            'BUSY': 'bg-rose-500',
            'EXAM-HALL DUTY': 'bg-purple-500',
            'LEAVE': 'bg-slate-400',
            'HALF-DAY PERMISSION': 'bg-orange-500'
        };

            <motion.div initial={{opacity:0}} animate={{opacity:1}} className="max-w-[1400px] mx-auto space-y-8">

                        <div>
                            <p className="text-[10px] font-black uppercase tracking-widest text-indigo-400 mb-1">My Current Availability</p>

                        <select
